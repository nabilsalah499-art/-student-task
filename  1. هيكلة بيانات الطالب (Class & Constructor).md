// 1. **هيكلة** **بيانات** **الطالب** **(**Class & Constructor**)**  
class Student {  
    constructor**(**name, id, grades**)** {  
        this.name = name;  
        this.id = id;  
        this.grades = grades;  
    }  
  
    // 2. **معالجة** **الدرجات** **(**Methods & Functions**)**  
    applyGraceMarks**()** {  
        this.grades = this.grades.map**(**grade => **(**grade === 48 || grade === 49**)** ? 50 **:** grade**)**;  
    }  
  
    analyzeGrades**()** {  
        const totalSum = this.grades.reduce**((**acc, curr**)** => acc + curr, 0**)**;  
        const maxGrade = this.grades.reduce**((**max, curr**)** => Math.max**(**max, curr**)**, this.grades[0]**)**;  
        const minGrade = this.grades.reduce**((**min, curr**)** => Math.min**(**min, curr**)**, this.grades[0]**)**;  
  
        return {  
            totalSum**:** totalSum,  
            maxGrade**:** maxGrade,  
            minGrade**:** minGrade  
        };  
    }  
}  
  
// 3. **تشغيل** **النظام** & Objects **(**forEach**)**  
let universitySystem = [];  
  
const student1 = new Student**("أحمد** **محمد"**, 101, [48, 85, 90]**)**;   
const student2 = new Student**("سارة** **أحمد"**, 102, [40, 45, 30]**)**;   
const student3 = new Student**("محمود** **خالد"**, 103, [95, 90, 88]**)**;   
  
const studentsList = [student1, student2, student3];  
  
studentsList.forEach**(**student => {  
    student.applyGraceMarks**()**;   
    universitySystem.push**(**student**)**;  
}**)**;  
  
// 4. **الفرز** **المتقدم** **(**Advanced Filtering**)**  
const passedStudents = universitySystem.filter**(**student => {  
    const passedGradesCount = student.grades.filter**(**grade => grade >= 50**)**.length;  
    return passedGradesCount === student.grades.length;  
}**)**;  
  
const failedStudents = universitySystem.filter**(**student => {  
    const failedGradesCount = student.grades.filter**(**grade => grade < 50**)**.length;  
    return failedGradesCount > 0;  
}**)**;  
  
// 5. **المخرجات** **(**Console Logging**)**  
console.log**("**--- **المصفوفة** **الشاملة** **لجميع** **الطلاب** **بعد** **معالجة** **الدرجات** ---**")**;  
console.log**(**universitySystem**)**;  
  
console.log**("**--- **قائمة** **الطلاب** **الناجحين** **بوضوح** ---**")**;  
console.log**(**passedStudents**)**;  
  
console.log**("**--- **قائمة** **الطلاب** **الراسبين** **بوضوح** ---**")**;  
console.log**(**failedStudents**)**;
